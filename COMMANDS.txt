--------------------------------------------------------------
  FULL PIPELINE (end-to-end)
--------------------------------------------------------------

# Step 1 — Generate regression coefficients from MODTRAN 3.7 data
#   Output: coefficients/unfiltering_coefficients_v*.nc
python -c "
from prod.std.standard_method import run
run(
    data_dir='data/Modtran_3-7_data/',
    srf_dir='data/SRF/',
    srf_version='0-0-1',
    modtran_version='3.7',
)
"

# Step 2 — Run the L2 unfiltering algorithm
#   Reads:  L1B and FMATCH-CAM files listed in the manifest
#   Writes: LIBERA_L2_UNF-RAD_*.nc  and  LIBERA_OUTPUT_MANIFEST_*.json
#           into $PROCESSING_PATH
export PROCESSING_PATH=/tmp/libera_output
mkdir -p $PROCESSING_PATH

.venv/bin/python unfiltered_radiances/algorithm.py data/manifest/LIBERA_INPUT_MANIFEST_01KVRA2GC8R3NGRTE5J8X2P8PE.json

# Step 3 — Inspect the output
python -c "
import xarray as xr, glob, numpy as np
f = sorted(glob.glob('/tmp/libera_output/LIBERA_L2_*.nc'))[-1]
print('Output file:', f)
ds = xr.open_dataset(f)
print(ds)
for v in ['shortwave_unfiltered_radiance', 'longwave_unfiltered_radiance',
          'split_shortwave_unfiltered_radiance', 'total_unfiltered_radiance']:
    a = ds[v].values
    print(f'{v}:  filled={np.isfinite(a).sum()}  nan={np.isnan(a).sum()}')
ds.close()
"

--------------------------------------------------------------
  GENERATE THE TWO-FILE MANIFEST (one-time setup)
--------------------------------------------------------------

# Run this if you need to recreate the manifest that includes both
# the L1B and FMATCH-CAM example files.
python -c "
from pathlib import Path
from libera_utils.io.manifest import Manifest, ManifestType

repo = Path('.').resolve()
filenames = [
    str(repo / 'data/example_data/LIBERA_L1B_RAD-4CH_V0-5-5_20280212T033945_20280212T052007_R26154153835.nc'),
    str(repo / 'data/example_data/LIBERA_ANC_FMATCH-CAM_V0-1-0_20280212T033945_20280212T033954_R26169165032.nc'),
]
manifest = Manifest(manifest_type=ManifestType.INPUT, files=filenames)
out = manifest.write(str(repo / 'data/manifest/'))
print('Written:', out)
"

--------------------------------------------------------------
  NOTES
--------------------------------------------------------------
- Two-file manifest (L1B + FMATCH-CAM):
    data/manifest/LIBERA_INPUT_MANIFEST_01KVRA2GC8R3NGRTE5J8X2P8PE.json
- Coefficient files live in coefficients/ and are named:
    unfiltering_coefficients_v{semver}_srf-{srf_ver}_modtran-{m_ver}.nc
- All prod commands use .venv (Poetry), NOT conda mr_env
- conda mr_env is only for research/notebooks/
- PROCESSING_PATH env var must be set before running algorithm.py
- COEFFICIENTS_FILE env var overrides the default coefficient file lookup:
    export COEFFICIENTS_FILE=/path/to/custom_coef.nc
